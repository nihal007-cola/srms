## 📝 **Step 6: Backend Code (main.py)**

```python
# main.py - FastAPI Backend
from fastapi import FastAPI, Depends, HTTPException
from fastapi.middleware.cors import CORSMiddleware
from sqlalchemy.orm import Session
import models, schemas, crud, auth, reports
from database import get_db

app = FastAPI(title="ERP Module API")

# CORS - Allow frontend
app.add_middleware(
    CORSMiddleware,
    allow_origins=["https://yourdomain.github.io"],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

# ============ AUTH ENDPOINTS ============
@app.post("/api/auth/login")
def login(user: schemas.UserLogin, db: Session = Depends(get_db)):
    return auth.authenticate_user(db, user)

@app.post("/api/auth/register")
def register(user: schemas.UserCreate, db: Session = Depends(get_db)):
    return auth.create_user(db, user)

# ============ ERP MODULE ENDPOINTS ============
@app.get("/api/erp/inventory")
def get_inventory(current_user = Depends(auth.get_current_user), db: Session = Depends(get_db)):
    return crud.get_inventory(db, current_user.company_id)

@app.post("/api/erp/orders")
def create_order(order: schemas.OrderCreate, current_user = Depends(auth.get_current_user), db: Session = Depends(get_db)):
    return crud.create_order(db, order, current_user.id)

@app.get("/api/erp/reports/daily")
def get_daily_report(current_user = Depends(auth.get_current_user), db: Session = Depends(get_db)):
    return crud.get_daily_report(db, current_user.company_id)

# ============ GOOGLE SHEETS REPORTING ============
@app.post("/api/reports/send")
def send_report(report_data: dict, current_user = Depends(auth.get_current_user)):
    return reports.send_to_google_sheets(report_data)

# ============ CRON JOB (Run daily) ============
@app.get("/api/cron/daily-report")
def daily_report_job():
    # This endpoint is called by cron
    data = reports.generate_daily_report()
    reports.send_to_google_sheets(data)
    return {"status": "Report sent"}
```

## 📊 **Step 7: Google Sheets Reporting (reports.py)**

```python
# reports.py - Google Sheets Integration
import google.auth
from google.oauth2 import service_account
from googleapiclient.discovery import build
import json
import os

SCOPES = ['https://www.googleapis.com/auth/spreadsheets']
SERVICE_ACCOUNT_FILE = os.getenv('GOOGLE_SHEETS_CREDENTIALS')
SPREADSHEET_ID = os.getenv('REPORT_SHEET_ID')

def send_to_google_sheets(data):
    """Send report data to Google Sheets"""
    try:
        credentials = service_account.Credentials.from_service_account_file(
            SERVICE_ACCOUNT_FILE, scopes=SCOPES
        )
        
        service = build('sheets', 'v4', credentials=credentials)
        sheet = service.spreadsheets()
        
        # Format data for Google Sheets
        values = [
            ['Date', 'Total Orders', 'Revenue', 'Customers'],
            [
                data['date'],
                data['total_orders'],
                data['revenue'],
                data['customers']
            ]
        ]
        
        body = {'values': values}
        result = sheet.values().update(
            spreadsheetId=SPREADSHEET_ID,
            range='Daily!A1',
            valueInputOption='USER_ENTERED',
            body=body
        ).execute()
        
        return {"status": "success", "updated": result.get('updatedCells')}
    except Exception as e:
        return {"status": "error", "message": str(e)}

def generate_daily_report():
    """Generate daily report from database"""
    # Query database for today's data
    # Return formatted report
    return {
        'date': '2026-07-23',
        'total_orders': 45,
        'revenue': 125000,
        'customers': 12
    }
```

```javascript
// Client-side routing - No .html in URLs
const routes = {
    '/': 'dashboard',
    '/inventory': 'inventory',
    '/orders': 'orders',
    '/reports': 'reports'
};

function navigate(path) {
    window.history.pushState({}, '', path);
    renderPage(routes[path] || '404');
}

function renderPage(page) {
    // Load page content
    fetch(`/api/pages/${page}`)
        .then(response => response.text())
        .then(html => {
            document.getElementById('app').innerHTML = html;
        });
}

// Handle browser back/forward
window.addEventListener('popstate', () => {
    renderPage(routes[window.location.pathname] || '404');
});

// Initial load
document.addEventListener('DOMContentLoaded', () => {
    const path = window.location.pathname;
    renderPage(routes[path] || 'dashboard');
});
```

### **api.js:**
```javascript
// API calls to backend
const API_URL = 'https://erp-api.yourdomain.com';

async function apiCall(endpoint, method = 'GET', data = null) {
    const response = await fetch(`${API_URL}${endpoint}`, {
        method: method,
        headers: {
            'Content-Type': 'application/json',
            'Authorization': `Bearer ${localStorage.getItem('token')}`
        },
        body: data ? JSON.stringify(data) : null
    });
    return response.json();
}

// Example: Get inventory
async function getInventory() {
    return await apiCall('/api/erp/inventory');
}
```
